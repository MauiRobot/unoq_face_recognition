async function retakeFace() { retakeButton.disabled = true; retakeButton.textContent = "Retaking..."; showFeedback( enrollmentStatus, "Capturing a replacement face..." ); try { const response = await fetch( "/capture", { method: "POST", } ); const result = await response.json(); if (!response.ok || !result.success) { throw new Error( result.message || "Unable to retake the face." ); } capturedPreview.src = result.preview_url + "?v=" + Date.now(); capturedPreview.hidden = false; previewPlaceholder.hidden = true; showFeedback( captureStatus, "✓ Replacement face captured.", "success" ); showFeedback( enrollmentStatus, "✓ Preview replaced. Review it before saving.", "success" ); await loadEvents(); } catch (error) { showFeedback( enrollmentStatus, "Retake failed: " + error.message, "error" ); } finally { retakeButton.disabled = false; retakeButton.textContent = "Retake"; } }
